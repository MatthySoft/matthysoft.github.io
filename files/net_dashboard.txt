  <init>
    <!-- nets / ign always get a dummy 192.0.2.0/32 entry (reserved TEST-NET address, never real traffic)
         so they are multivalue even when the box holds a single CIDR; mvmap needs that.
         Shared enrichment, used as  | eval nets=mvappend("192.0.2.0/32=unused", split($nets|s$, ",")), ign=mvappend("192.0.2.0/32", split($ignore|s$, ",")) | $enrich$
         Drops anything to/from the Ignore subnets.
         Adds: src_ip dest_ip dport, src_net dest_net (label, empty = external),
               src / dest ("IP (label)"), direction (outbound|inbound|internal|external), dir.
         Multicast / broadcast / link-local destinations count as internal. -->
    <set token="enrich">rename id.orig_h AS src_ip, id.resp_h AS dest_ip, id.resp_p AS dport
| eval ignored=mvindex(mvmap(ign, if(cidrmatch(if(match(ign, "/"), trim(ign), "192.0.2.0/32"), src_ip) OR cidrmatch(if(match(ign, "/"), trim(ign), "192.0.2.0/32"), dest_ip), 1, null())), 0)
| where isnull(ignored)
| eval src_net=mvindex(mvmap(nets, if(cidrmatch(trim(mvindex(split(nets, "="), 0)), src_ip), coalesce(trim(mvindex(split(nets, "="), 1)), "internal"), null())), 0)
| eval dest_net=mvindex(mvmap(nets, if(cidrmatch(trim(mvindex(split(nets, "="), 0)), dest_ip), coalesce(trim(mvindex(split(nets, "="), 1)), "internal"), null())), 0)
| eval direction=case(
    isnotnull(src_net) AND isnotnull(dest_net), "internal",
    cidrmatch("224.0.0.0/4", dest_ip) OR cidrmatch("169.254.0.0/16", dest_ip) OR dest_ip="255.255.255.255" OR match(dest_ip, "(?i)^(ff0|fe80:)"), "internal",
    isnotnull(src_net), "outbound",
    isnotnull(dest_net), "inbound",
    true(), "external")
| eval src=if(isnotnull(src_net), src_ip." (".src_net.")", src_ip),
       dest=if(isnotnull(dest_net), dest_ip." (".dest_net.")", dest_ip),
       dir=case(direction="outbound", "OUT -&gt;", direction="inbound", "&lt;- IN", direction="internal", "INT", true(), "EXT")
| fields - nets ign ignored</set>
    <set token="prev10">...</set>

  <search id="sus_conn">
    <query>index=zeek sourcetype="bro:conn:json" $search$
| eval nets=mvappend("192.0.2.0/32=unused", split($nets|s$, ",")), ign=mvappend("192.0.2.0/32", split($ignore|s$, ","))
| $enrich$
| search $direction$
| fillnull value="-" service
| eval reason=case(
    in(dport, 4444, 1337, 31337, 5555), "Known backdoor / C2 port",
    in(dport, 9001, 9030, 9050, 9150), "Tor port",
    match(service, "irc") OR in(dport, 6660, 6666, 6667, 6697), "IRC",
    match(service, "socks") OR dport=1080, "SOCKS proxy",
    match(service, "(telnet|rlogin|ftp)") OR in(dport, 21, 23), "Cleartext login protocol",
    match(service, "(smb|dce_rpc|ntlm|krb|ldap|rdp|rfb)") OR in(dport, 135, 139, 445, 3389, 5900, 5985, 5986), "Windows / remote-admin protocol",
    service="ssh" AND dport!=22, "SSH on non-standard port",
    service="http" AND NOT in(dport, 80, 8080, 8000, 8008, 8888), "HTTP on non-standard port",
    service="ssl" AND NOT in(dport, 443, 8443, 993, 995, 465, 636, 853, 5986), "TLS on non-standard port",
    service="dns" AND dport!=53, "DNS on non-standard port",
    dport=53 AND service!="dns" AND conn_state="SF", "Non-DNS traffic on port 53",
    proto="tcp" AND in(dport, 80, 443) AND service="-" AND conn_state="SF" AND resp_bytes&gt;0, "Unidentified traffic on web port")
| where isnotnull(reason)
| eval protocol=if(service="-", proto." (unidentified)", service), port=proto."/".dport
| stats count AS conns latest(_time) AS last_seen BY reason protocol port src dest
| sort - last_seen
| eval last_seen=strftime(last_seen, "%Y-%m-%d %H:%M:%S")
| table last_seen reason protocol port src dest conns</query>

  <search id="brute">
    <query>index=zeek sourcetype="bro:conn:json" $search$ id.resp_p IN (21, 22, 23, 88, 139, 389, 445, 636, 1433, 3306, 3389, 5432, 5900, 5985, 5986)
| eval nets=mvappend("192.0.2.0/32=unused", split($nets|s$, ",")), ign=mvappend("192.0.2.0/32", split($ignore|s$, ","))
| $enrich$
| search $direction$
| eval port_name=case(dport=21,"FTP", dport=22,"SSH", dport=23,"Telnet", dport=88,"Kerberos", dport=139,"NetBIOS", dport=389,"LDAP", dport=445,"SMB", dport=636,"LDAPS", dport=1433,"MSSQL", dport=3306,"MySQL", dport=3389,"RDP", dport=5432,"PostgreSQL", dport=5900,"VNC", dport=5985,"WinRM", dport=5986,"WinRM-TLS")
| bin _time span=10m
| stats count AS attempts values(conn_state) AS states BY _time dir src dest dport port_name
| where attempts &gt;= 20
| stats max(attempts) AS peak_per_10m sum(attempts) AS total max(_time) AS last_window values(states) AS states BY dir src dest dport port_name
| sort - peak_per_10m
| eval last_window=strftime(last_window, "%Y-%m-%d %H:%M"), port="tcp/".dport
| rename port_name AS protocol
| table last_window protocol port src dest peak_per_10m total</query>
    <earliest>$range$</earliest>
    <latest>now</latest>
    <done><set token="n_brute">$job.resultCount$</set></done>
